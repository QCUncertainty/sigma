510// The bounds are computed first and the nominal value second. interval_t's
511// operations restore the rounding mode before they return, so the nominal value
512// is computed in round-to-nearest, as plain floating point would compute it.

514template<typename IntervalType>
515Triplex<IntervalType>& Triplex<IntervalType>::operator+=(const Triplex& rhs) {
516 if(empty()) { return *this = rhs; }
517 if(rhs.empty()) { return *this; }
518 auto bounds = m_interval_ + rhs.m_interval_;
519 value_t nominal = m_nominal_ + rhs.m_nominal_;
520 return *this = detail_::make_triplex_(std::move(bounds), nominal);
521}

643template<typename IntervalType>
644bool operator<(const Triplex<IntervalType>& lhs,
645 const Triplex<IntervalType>& rhs) {
646 if(rhs.empty()) { return false; }
647 if(lhs.empty()) { return true; }
648 const auto key = [](const Triplex<IntervalType>& t) {
649 bool is_left_open = false;
650 bool is_right_open = false;
651 if constexpr(is_general_interval_v<IntervalType>) {
652 is_left_open = t.left_open();
653 is_right_open = t.right_open();
654 }
655 return std::make_tuple(t.nominal(), t.lower(), t.upper(), is_left_open,
656 is_right_open);
657 };
658 return key(lhs) < key(rhs);
659}
