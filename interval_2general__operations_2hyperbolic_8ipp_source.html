16template<typename T>
17GeneralInterval<T> cosh(const GeneralInterval<T>& a) {
18 if(a.empty()) { return GeneralInterval<T>(); }
19
20 bool left_open = a.left_open();
21 bool right_open = a.right_open();
22 if(a.upper() <= T(0)) {
23 // Entirely non-positive, where cosh is decreasing. A zero upper bound
24 // is an input bound, attained only if it is closed.
25 std::swap(left_open, right_open);
26 } else if(a.lower() < T(0)) {
27 // Spans zero, so the minimum is the interior point cosh(0) == 1 and
28 // the maximum comes from whichever bound is farther from zero.
29 const T abs_lower = -a.lower();
30 left_open = false;
31 if(abs_lower > a.upper()) {
32 right_open = a.left_open();
33 } else if(abs_lower < a.upper()) {
34 right_open = a.right_open();
35 } else {
36 right_open = a.left_open() && a.right_open();
37 }
38 }
39
40 return detail_::general_from_boost_(
41 boost::numeric::cosh(detail_::to_boost_(a)), left_open, right_open);
42}

60template<typename T>
61GeneralInterval<T> acosh(const GeneralInterval<T>& a) {
62 if(a.empty()) { return GeneralInterval<T>(); }
63 // Trimming the argument back to the domain would answer a narrower
64 // question than the one asked. Boost would clamp the lower bound to one.
65 if(a.lower() < T(1)) {
66 throw std::domain_error("Interval has values less than 1.");
67 }
68 // Increasing, so the openness of each bound carries over.
69 return detail_::general_from_boost_(
70 boost::numeric::acosh(detail_::to_boost_(a)), a.left_open(),
71 a.right_open());
72}

74template<typename T>
75GeneralInterval<T> atanh(const GeneralInterval<T>& a) {
76 if(a.empty()) { return GeneralInterval<T>(); }
77 // The domain is open, so a bound AT plus or minus one is out of domain
78 // only if it is closed -- the same rule log() applies at zero.
79 detail_::assert_within_unit_(a, true, "inverse hyperbolic tangent");
80 // Increasing. An open bound at either end of the domain leaves the result
81 // unbounded on that side, and from_boost_ opens that infinite bound.
82 return detail_::general_from_boost_(
83 boost::numeric::atanh(detail_::to_boost_(a)), a.left_open(),
84 a.right_open());
85}
