388template<typename ValueType>
389bool Interval<ValueType>::contains(value_t value) const {
390 if(empty()) { return false; }
391
392 // Check if the value is definitely outside the interval
393 if(value < lower() || value > upper()) { return false; }
394
395 // Now we know it's squarely in the interval (or one of the bounds)
396
397 // Not allowed to be a bound if the interval is open on that side
398 if(left_open() && value == lower()) { return false; }
399 if(right_open() && value == upper()) { return false; }
400
401 return true;
402}

404template<typename ValueType>
405bool Interval<ValueType>::contains(const Interval& other) const {
406 if(other.empty()) { return true; }
407 if(empty()) { return false; }
408
409 // Check if definitely outside the interval
410 if(other.lower() < lower() || other.upper() > upper()) { return false; }
411
412 // Check if the interval is squarely inside the interval
413 if(other.lower() > lower() && other.upper() < upper()) { return true; }
414
415 // *this and other must share at least one bound
416 // For each bound the choices are:
417 // *this open and other closed -> not okay
418 // *this open and other open -> okay
419 // *this closed and other closed -> okay
420 // *this closed and other open -> okay
421 if(other.lower() == lower()) {
422 if(left_open() && other.left_closed()) { return false; }
423 }
424 if(other.upper() == upper()) {
425 if(right_open() && other.right_closed()) { return false; }
426 }
427 return true;
428}

430template<typename ValueType>
431Interval<ValueType> Interval<ValueType>::set_union(
432 const Interval& other) const {
433 if(empty()) { return other; }
434 if(other.empty()) { return *this; }
435 if(set_intersection(other).empty()) {
436 throw std::domain_error("Intervals do not overlap");
437 }
438 value_t low = lower();
439 value_t hi = upper();
440 bool lower_is_open = false;
441 bool upper_is_open = false;
442 if(lower() < other.lower()) { // *this has the lower bound
443 lower_is_open = left_open();
444 } else if(lower() == other.lower()) {
445 // If both are open, then result is open; otherwise closed
446 lower_is_open = left_open() && other.left_open();
447 } else {
448 low = other.lower();
449 lower_is_open = other.left_open();
450 }
451
452 if(upper() > other.upper()) {
453 upper_is_open = right_open();
454 } else if(upper() == other.upper()) {
455 upper_is_open = right_open() && other.right_open();
456 } else {
457 hi = other.upper();
458 upper_is_open = other.right_open();
459 }
460
461 return Interval(low, hi, lower_is_open, upper_is_open);
462}

512template<typename ValueType>
513Interval<ValueType>& Interval<ValueType>::operator*=(const Interval& rhs) {
514 /* Putting this here in case it's helpful for optimizing.
515 * Given two intervals X = [a, b] and Y = [c, d], the product X * Y is:
516 * [min(ac, ad, bc, bd), max(ac, ad, bc, bd)]. Thus the entire product
517 * is determined by four sub-products: ac, ad, bc, and bd. Simple
518 * combinatorics suggests that there are 16 possible products for X*Y,
519 * (i.e., chose one sub-product for the new lower bound and one for the
520 * new upper bound).
521 *
522 * If we know that a < b and c < d, i.e., our intervals are not points,
523 * we can immediately rule out eight products:
524 *
525 * - [ac, ac], [ad, ad], [bc, bc], and [bd,bd].
526 * - [ac, ad], [ac, bc], [bd, bc], and [bd,ad].
527 *
528 * Proofs that these products are impossible are given below.
529 *
530 * To prove the first four products are impossible consider [ac, ac],
531 * which implies:
532 *
533 * - ac <= ad <= ac,
534 * - ac <= bc <= ac, and
535 * -ac <= bd <= ac
536 *
537 * The first inequality means c == d, the second means a == b. Which
538 * means X = [a, a] and Y = [c, c], i.e. X and Y must be points. This
539 * contridicts our assumption that X and Y are not points and thus these
540 * four products will never appear. Similar logic holds for the other
541 * three products of the form [xy, xy].
542 *
543 * Now consider the product [ac, ad]. This implies:
544 *
545 * - ac <= bc <= ad, and
546 * - ac <= bd <= ad
547 *
548 * If a is negative, then d <= bc/a <= c, which is not possible with the
549 * assumption c < d. If a is zero, the product [ac, ad] is [0, 0], which
550 * means bc and bd must be zero. b can't be zero because then
551 * X = [0, 0]. This would then mean that both c and d must be zero, but
552 * c and d both can't be zero because then Y = [0, 0]. Thus a must be
553 * positive. If a is positive, then so is b. If c >= 0 then d is
554 * positive and then we have that bd > ad, which is not consistent with
555 * bd <= ad. c also can't be negative because then bc < ac. We thus
556 * conclude that [ac, ad] is not possible.
557 *
558 * The remaining three proofs follow similarly. Let "target interval"
559 * refer to the interval we are proving is not possible. Then the proof
560 * proceeds by showing:
561 *
562 * - We get two inequalities by forcing the remaining two sub-products
563 * to be bounded by the target interval. Above this showed that bc and
564 * bd must be in the interval [ac, ad]. We term these the "implicit
565 * sub-products" because they do not show up explicitly in the target
566 * interval.
567 * - One of X or Y's bounds will appear twice in the target interval. We
568 * term this the "common bound". Above this bound was "a".
569 * - If the common bound is the lower bound of X(Y) try assuming the
570 * commonbound is negative and dividing the inequalities by it. If the
571 * common bound is the upper bound, instead divide the inequalities by
572 * the positive of it. This will cause one of the two inequalities to
573 * be a contradiction eliminating that possibility. Above this ruled
574 * out "a" being negative.
575 * - The common bound can't be zero otherwise X and Y collapse to
576 * point intervals.
577 * - This means the common bound must be positive (negative) for a lower
578 * (upper) bound. In turn, the "other bound" must be positive
579 * (negative) too. Above this "other bound" was "b".
580 * - The implicit sub-products will share other bound. If other bound is
581 * the upper (lower) bound, then assume that the lower (upper) bound
582 * of the other interval is positive (negative) or zero. This
583 * will force the upper (lower) bound of the other interval to be
584 * positive (negative) too. All bounds will now have the same sign
585 * and violate one of the inequalities.
586 * - The last possibility is then that the lower (upper) bound of the
587 * other interval has the opposite sign as the common bound. This too
588 * will violate one of the inequalities and we conclude that the
589 * target interval is not possible.
590 */
591
592 if(empty() || rhs.empty()) { return *this = Interval(); }
593
594 // The bounds themselves come from boost, which rounds each of the four
595 // sub-products outward before taking the extremes. Recomputing the
596 // sub-products here in the ambient rounding mode is only to identify which
597 // corner each bound came from, which is what decides its openness; a
598 // corner that ties with another to within a rounding is interchangeable
599 // with it for that purpose.
600 auto product = *m_interval_ * *rhs.m_interval_;
601
602 auto ll = lower() * rhs.lower();
603 auto lh = lower() * rhs.upper();
604 auto hl = upper() * rhs.lower();
605 auto hh = upper() * rhs.upper();
606 auto low = std::min(std::min(ll, lh), std::min(hl, hh));
607 auto high = std::max(std::max(ll, lh), std::max(hl, hh));
608
609 auto [lower_is_open, upper_is_open] =
610 corner_openness_(low, high, ll, lh, hl, hh, *this, rhs);
611
612 return *this = Interval(product.lower(), product.upper(), lower_is_open,
613 upper_is_open);
614}

616template<typename ValueType>
617std::pair<bool, bool> Interval<ValueType>::corner_openness_(
618 value_t low, value_t high, value_t ll, value_t lh, value_t hl, value_t hh,
619 const Interval& lhs, const Interval& rhs) {
620 bool lower_is_open = lhs.left_open() || rhs.left_open();
621 bool upper_is_open = lhs.right_open() || rhs.right_open();
622
623 // if(low == ll) // default is correct for this case
624 if(low == lh) {
625 lower_is_open = lhs.left_open() || rhs.right_open();
626 } else if(low == hl) {
627 lower_is_open = lhs.right_open() || rhs.left_open();
628 } else if(low == hh) {
629 lower_is_open = lhs.right_open() || rhs.right_open();
630 }
631
632 // if(high == hh) // default is correct for this case
633 if(high == lh) {
634 upper_is_open = lhs.left_open() || rhs.right_open();
635 } else if(high == hl) {
636 upper_is_open = lhs.right_open() || rhs.left_open();
637 } else if(high == ll) {
638 upper_is_open = lhs.left_open() || rhs.left_open();
639 }
640
641 return {lower_is_open, upper_is_open};
642}

644template<typename ValueType>
645Interval<ValueType> Interval<ValueType>::reciprocal_(const Interval& a) {
646 constexpr auto inf = std::numeric_limits<value_t>::infinity();
647 const interval_t one(value_t(1));
648
649 // A zero bound is necessarily an open one here, and the reciprocal runs
650 // off to infinity in that direction. Everything else is a directed
651 // division by a point.
652 value_t low =
653 a.upper() == value_t(0) ? -inf : (one / interval_t(a.upper())).lower();
654 value_t high =
655 a.lower() == value_t(0) ? inf : (one / interval_t(a.lower())).upper();
656
657 // The reciprocal is decreasing, so the bounds trade places
658 return Interval(low, high, a.right_open(), a.left_open());
659}

661template<typename ValueType>
662Interval<ValueType>& Interval<ValueType>::operator/=(const Interval& rhs) {
663 if(empty() || rhs.empty()) { return *this = Interval(); }
664 if(rhs.contains(value_t(0))) {
665 throw std::domain_error("Division by zero");
666 }
667
668 // Boost has no notion of an open bound, so a divisor bounded by zero looks
669 // to it like a divisor containing zero and it answers with the whole real
670 // line. Sigma knows better, and multiplying by the reciprocal keeps the
671 // information at the cost of one extra rounding.
672 if(rhs.lower() == value_t(0) || rhs.upper() == value_t(0)) {
673 return *this *= reciprocal_(rhs);
674 }
675
676 // As in operator*=, boost computes the bounds and the quotients below only
677 // say which corner each of them came from.
678 auto quotient = *m_interval_ / *rhs.m_interval_;
679
680 auto ll = lower() / rhs.lower();
681 auto lh = lower() / rhs.upper();
682 auto hl = upper() / rhs.lower();
683 auto hh = upper() / rhs.upper();
684 auto low = std::min(std::min(ll, lh), std::min(hl, hh));
685 auto high = std::max(std::max(ll, lh), std::max(hl, hh));
686
687 auto [lower_is_open, upper_is_open] =
688 corner_openness_(low, high, ll, lh, hl, hh, *this, rhs);
689
690 return *this = Interval(quotient.lower(), quotient.upper(), lower_is_open,
691 upper_is_open);
692}

738template<typename T1, typename T2>
739bool operator==(const Interval<T1>& lhs, const Interval<T2>& rhs) {
740 if constexpr(!std::is_same_v<T1, T2>) return false;
741 if(lhs.empty() && rhs.empty()) { return true; }
742 if(lhs.empty() || rhs.empty()) { return false; }
743 if(lhs.left_open() != rhs.left_open()) { return false; }
744 if(lhs.right_open() != rhs.right_open()) { return false; }
745 return lhs.lower() == rhs.lower() && lhs.upper() == rhs.upper();
746}
