31template<typename T, typename U>
32Interval<T> pow(const Interval<T>& a, const U& exp) {
33 if(a.empty()) { return Interval<T>(); }
34 if(exp == U(0)) { return Interval<T>(T(1), T(1)); }
35
36 // x^exp is monotonic across [a.lower(), a.upper()] -- and therefore
37 // attains its extrema at the endpoints, which is all the code below
38 // checks -- EXCEPT when a spans 0 and exp is a positive even integer: x^2
39 // (for example) decreases from a.lower() down to 0 and then increases
40 // back up to a.upper(), so 0 (an interior point, not either endpoint) is
41 // the true minimum. Handle that case directly; every other combination
42 // of sign and exponent is already monotonic end-to-end.
43 bool spans_zero = a.lower() < T(0) && a.upper() > T(0);
44 bool is_integer = std::floor(exp) == exp;
45 if(spans_zero && exp > U(0) && is_integer &&
46 static_cast<long long>(exp) % 2 == 0) {
47 auto abs_lo = -a.lower();
48 auto abs_hi = a.upper();
49 if(abs_lo > abs_hi) {
50 return Interval<T>(T(0), std::pow(abs_lo, exp), false,
51 a.left_open());
52 }
53 if(abs_hi > abs_lo) {
54 return Interval<T>(T(0), std::pow(abs_hi, exp), false,
55 a.right_open());
56 }
57 return Interval<T>(T(0), std::pow(abs_hi, exp), false,
58 a.left_open() && a.right_open());
59 }
60
61 auto low = std::pow(a.lower(), exp);
62 auto high = std::pow(a.upper(), exp);
63 if(low > high) {
64 return Interval<T>(high, low, a.right_open(), a.left_open());
65 }
66 return Interval<T>(low, high, a.left_open(), a.right_open());
67}
