27template<typename T, typename U>
28boost_interval_t<T> exponent_interval_(const U& exp) {
29 const auto value = static_cast<T>(exp);
30 if(static_cast<U>(value) == exp) { return boost_interval_t<T>(value); }

32 return boost_interval_t<T>(std::nextafter(value, -inf),

67template<typename T, typename U>
68Interval<T> pow(const Interval<T>& a, const U& exp) {
69 if(a.empty()) { return Interval<T>(); }
70 if(exp == U(0)) { return Interval<T>(T(1), T(1)); }
71
72 // A negative exponent is a reciprocal, and dividing by an interval that
73 // contains zero is the error operator/= already rejects.
74 if(exp < U(0) && a.contains(T(0))) {
75 throw std::domain_error(
76 "Can not raise an interval containing 0 to a negative power.");
77 }
78
79 auto x = detail_::to_boost_(a);
80
81 if(!detail::is_integer_exponent(exp)) {
82 // A negative base raised to a fractional power is not a real number,
83 // so the base must be non-negative. Zero itself is fine for a positive
84 // exponent: the evaluation below runs through log, whose unbounded
85 // result exponentiates back to the 0 that 0^exp really is.
86 if(a.lower() < T(0)) {
87 throw std::domain_error("Can not raise an interval with negative "
88 "values to a non-integer power.");
89 }
90 // Away from that, a non-integer power is monotonic: increasing for a
91 // positive exponent, decreasing for a negative one.
92 auto y = boost::numeric::exp(detail_::exponent_interval_<T>(exp) *
93 boost::numeric::log(x));
94 return detail_::from_boost_(y);
95 }
96
97 const auto power = static_cast<long long>(exp);
98 return detail_::from_boost_(
99 boost::numeric::pow(x, static_cast<int>(power)));
100}

102template<typename T>
103Interval<T> pow(const Interval<T>& a, const Interval<T>& exponent) {
104 if(a.empty() || exponent.empty()) { return Interval<T>(); }
105 // The point-exponent case is expected to be far more common than a
106 // genuinely uncertain exponent, so it is forwarded to the scalar
107 // overload for tighter bounds and that overload's more permissive
108 // domain restriction (e.g., a negative base raised to an integer power
109 // is fine, even though it is not a valid base for log()).
110 if(exponent.width() == T(0)) { return pow(a, exponent.lower()); }
111 // log() enforces the strictly-positive-base domain restriction and
112 // throws std::domain_error itself, so there's nothing left to check
113 // here.
114 return sigma::exp(exponent * sigma::log(a));
115}
