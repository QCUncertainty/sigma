29template<typename T>
30std::pair<bool, bool> integer_pow_openness_(const Interval<T>& a, long long n) {
31 const bool is_odd = (n % 2 != 0);
32 bool left_open = a.left_open();
33 bool right_open = a.right_open();
34
35 if(a.upper() < T(0)) {
36 // Entirely negative. An odd power is increasing, an even one is
37 // decreasing, which swaps which input bound each output bound came
38 // from.
39 if(!is_odd) { std::swap(left_open, right_open); }
40 } else if(a.lower() < T(0)) {
41 // Spans zero. An odd power is still increasing across all of it, but
42 // an even power dips to an interior minimum at x == 0 and takes its
43 // maximum at whichever endpoint is farther from zero.
44 if(!is_odd) {
45 const T abs_lower = -a.lower();
46 left_open = false; // 0 is attained, and is an interior
47 // point rather than an input bound
48 if(abs_lower > a.upper()) {
49 right_open = a.left_open();
50 } else if(abs_lower < a.upper()) {
51 right_open = a.right_open();
52 } else {
53 right_open = a.left_open() && a.right_open();
54 }
55 }
56 }
57 // Entirely non-negative: increasing, so the openness carries over as-is.
58
59 return {left_open, right_open};
60}

78template<typename T, typename U>
79boost_interval_t<T> exponent_interval_(const U& exp) {
80 const auto value = static_cast<T>(exp);
81 if(static_cast<U>(value) == exp) { return boost_interval_t<T>(value); }
82 constexpr auto inf = std::numeric_limits<T>::infinity();
83 return boost_interval_t<T>(std::nextafter(value, -inf),

89template<typename T>
90Interval<T> sqrt(const Interval<T>& a) {
91 if(a.empty()) { return Interval<T>(); }
92 // Silently dropping the part of a that is out of domain would answer a
93 // question that was not asked, so say so instead. Boost would clamp the
94 // lower bound to zero.
95 if(a.lower() < T(0)) {
96 throw std::domain_error("Interval has negative values.");
97 }
98 // Square root is increasing, so the openness of each bound carries over.
99 return detail_::from_boost_(boost::numeric::sqrt(detail_::to_boost_(a)),
100 a.left_open(), a.right_open());

110template<typename T>
111Interval<T> log(const Interval<T>& a) {
112 if(a.empty()) { return Interval<T>(); }
113 // The domain is the strictly positive reals, so a closed bound at zero is
114 // out of domain while an open one is not: (0, 1] contains no non-positive
115 // value even though its lower bound is zero. Boost would clamp instead.
116 if(a.lower() < T(0) || (a.lower() == T(0) && a.left_closed())) {
117 throw std::domain_error("Interval has non-positive values.");
118 }
119 // An argument bounded below by an open zero is unbounded below in the
120 // result, and from_boost_ opens that infinite bound for us. Logarithm is
121 // increasing, so the openness of each bound otherwise carries over.
122 return detail_::from_boost_(boost::numeric::log(detail_::to_boost_(a)),
123 a.left_open(), a.right_open());
124}

126template<typename T, typename U>
127Interval<T> pow(const Interval<T>& a, const U& exp) {
128 if(a.empty()) { return Interval<T>(); }
129 if(exp == U(0)) { return Interval<T>(T(1), T(1)); }
130
131 // A negative exponent is a reciprocal, and dividing by an interval that
132 // contains zero is the error operator/= already rejects.
133 if(exp < U(0) && a.contains(T(0))) {
134 throw std::domain_error(
135 "Can not raise an interval containing 0 to a negative power.");
136 }
137
138 auto x = detail_::to_boost_(a);
139
140 if(!detail::is_integer_exponent(exp)) {
141 // A negative base raised to a fractional power is not a real number,
142 // so the base must be non-negative. Zero itself is fine for a positive
143 // exponent: the evaluation below runs through log, whose unbounded
144 // result exponentiates back to the 0 that 0^exp really is.
145 if(a.lower() < T(0)) {
146 throw std::domain_error("Can not raise an interval with negative "
147 "values to a non-integer power.");
148 }
149 // Away from that, a non-integer power is monotonic: increasing for a
150 // positive exponent, decreasing for a negative one.
151 auto y = boost::numeric::exp(detail_::exponent_interval_<T>(exp) *
152 boost::numeric::log(x));
153 if(exp < U(0)) {
154 return detail_::from_boost_(y, a.right_open(), a.left_open());
155 }
156 return detail_::from_boost_(y, a.left_open(), a.right_open());
157 }
158
159 const auto power = static_cast<long long>(exp);
160 auto [left_open, right_open] =
161 detail_::integer_pow_openness_(a, std::abs(power));
162 // A negative power is the reciprocal of the positive one, which reverses
163 // the interval and so exchanges the openness of its bounds.
164 if(power < 0) { std::swap(left_open, right_open); }
165
166 return detail_::from_boost_(boost::numeric::pow(x, static_cast<int>(power)),
167 left_open, right_open);
168}

170template<typename T>
171Interval<T> pow(const Interval<T>& a, const Interval<T>& exponent) {
172 if(a.empty() || exponent.empty()) { return Interval<T>(); }
173 // The point-exponent case is expected to be far more common than a
174 // genuinely uncertain exponent, so it is forwarded to the scalar
175 // overload for tighter bounds and that overload's more permissive
176 // domain restriction (e.g., a negative base raised to an integer power
177 // is fine, even though it is not a valid base for log()).
178 if(exponent.width() == T(0)) { return pow(a, exponent.lower()); }
179 // log() enforces the strictly-positive-base domain restriction and
180 // throws std::domain_error itself, so there's nothing left to check
181 // here.
182 return sigma::exp(exponent * sigma::log(a));
183}
