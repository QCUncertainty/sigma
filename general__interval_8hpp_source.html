502template<typename ValueType>
503bool GeneralInterval<ValueType>::contains(const GeneralInterval& other) const {
504 if(other.empty()) { return true; }
505 if(!m_closure_.contains(other.m_closure_)) { return false; }
506
507 // *this and other may share a bound. For each shared bound the choices
508 // are:
509 // *this open and other closed -> not okay
510 // *this open and other open -> okay
511 // *this closed and other closed -> okay
512 // *this closed and other open -> okay
513 if(other.lower() == lower()) {
514 if(left_open() && other.left_closed()) { return false; }
515 }
516 if(other.upper() == upper()) {
517 if(right_open() && other.right_closed()) { return false; }
518 }
519 return true;
520}

522template<typename ValueType>
523GeneralInterval<ValueType> GeneralInterval<ValueType>::set_union(
524 const GeneralInterval& other) const {
525 if(empty()) { return other; }
526 if(other.empty()) { return *this; }
527 // The intervals need not share a point: [0, 1) and [1, 2] only touch, but
528 // their union is [0, 2]. What breaks the union is a gap, i.e., bounds that
529 // are separated or that coincide at a point neither interval contains.
530 const auto separated = [](const GeneralInterval& left,
531 const GeneralInterval& right) {
532 if(left.upper() < right.lower()) { return true; }
533 return left.upper() == right.lower() && left.right_open() &&
534 right.left_open();
535 };
536 if(separated(*this, other) || separated(other, *this)) {
537 throw std::domain_error("Intervals do not overlap");
538 }
539 value_t low = lower();
540 value_t hi = upper();
541 bool lower_is_open = false;
542 bool upper_is_open = false;
543 if(lower() < other.lower()) { // *this has the lower bound
544 lower_is_open = left_open();
545 } else if(lower() == other.lower()) {
546 // If both are open, then result is open; otherwise closed
547 lower_is_open = left_open() && other.left_open();
548 } else {
549 low = other.lower();
550 lower_is_open = other.left_open();
551 }
552
553 if(upper() > other.upper()) {
554 upper_is_open = right_open();
555 } else if(upper() == other.upper()) {
556 upper_is_open = right_open() && other.right_open();
557 } else {
558 hi = other.upper();
559 upper_is_open = other.right_open();
560 }
561
562 return GeneralInterval(low, hi, lower_is_open, upper_is_open);
563}

565template<typename ValueType>
566GeneralInterval<ValueType> GeneralInterval<ValueType>::set_intersection(
567 const GeneralInterval& other) const {

569 if(other.upper() < lower() || other.lower() > upper()) {

575 bool upper_is_open = right_open();

579 lower_is_open = other.left_open();

581 lower_is_open = left_open() || other.left_open();

593template<typename ValueType>
594GeneralInterval<ValueType>& GeneralInterval<ValueType>::operator+=(
595 const GeneralInterval& rhs) {
596 if(empty()) { return *this = rhs; }
597 if(rhs.empty()) { return *this; }

599 m_is_left_open_ = left_open() || rhs.left_open();
600 m_is_right_open_ = right_open() || rhs.right_open();
601 // The sum may have overflowed to an infinite bound
602 return *this = GeneralInterval(lower(), upper(), left_open(), right_open());

605template<typename ValueType>
606bool GeneralInterval<ValueType>::corner_bound_open_(
607 value_t value, value_t ll, value_t lh, value_t hl, value_t hh,
608 const GeneralInterval& lhs, const GeneralInterval& rhs) {
609 bool is_open = true;
610 if(value == ll) { is_open &= lhs.left_open() || rhs.left_open(); }
611 if(value == lh) { is_open &= lhs.left_open() || rhs.right_open(); }
612 if(value == hl) { is_open &= lhs.right_open() || rhs.left_open(); }
613 if(value == hh) { is_open &= lhs.right_open() || rhs.right_open(); }
614 return is_open;
615}

617template<typename ValueType>
618GeneralInterval<ValueType> GeneralInterval<ValueType>::from_corners_(
619 const closure_t& result, value_t ll, value_t lh, value_t hl, value_t hh,
620 const GeneralInterval& lhs, const GeneralInterval& rhs, bool zero_attained) {
621 // fmin and fmax skip the NaN of a zero times an infinite bound, which is
622 // not a corner the result can come from.
623 auto low = std::fmin(std::fmin(ll, lh), std::fmin(hl, hh));
624 auto high = std::fmax(std::fmax(ll, lh), std::fmax(hl, hh));
625
626 // An attained zero is checked against the result rather than the corners,
627 // which are all NaN for, e.g., [0, 0] * (-inf, inf).
628 bool lower_is_open = !(zero_attained && result.lower() == value_t(0)) &&
629 corner_bound_open_(low, ll, lh, hl, hh, lhs, rhs);
630 bool upper_is_open = !(zero_attained && result.upper() == value_t(0)) &&
631 corner_bound_open_(high, ll, lh, hl, hh, lhs, rhs);
632
633 return GeneralInterval(result.lower(), result.upper(), lower_is_open,
634 upper_is_open);
635}

650 return *this =

654template<typename ValueType>
655GeneralInterval<ValueType> GeneralInterval<ValueType>::reciprocal_(
656 const GeneralInterval& a) {
657 using interval_t = detail_::boost_interval_t<value_t>;
658 constexpr auto inf = std::numeric_limits<value_t>::infinity();
659 const interval_t one(value_t(1));
660
661 // A zero bound is necessarily an open one here, and the reciprocal runs
662 // off to infinity in that direction. Everything else is a directed
663 // division by a point.
664 value_t low =
665 a.upper() == value_t(0) ? -inf : (one / interval_t(a.upper())).lower();
666 value_t high =
667 a.lower() == value_t(0) ? inf : (one / interval_t(a.lower())).upper();
668
669 // The reciprocal is decreasing, so the bounds trade places
670 return GeneralInterval(low, high, a.right_open(), a.left_open());
671}

673template<typename ValueType>
674GeneralInterval<ValueType>& GeneralInterval<ValueType>::operator/=(
675 const GeneralInterval& rhs) {
676 if(empty() || rhs.empty()) { return *this = GeneralInterval(); }
677 if(rhs.contains(value_t(0))) {
678 throw std::domain_error("Division by zero");
679 }
680
681 // The closure of a divisor bounded by an open zero contains zero, so
682 // closed division would reject it. Multiplying by the reciprocal keeps the
683 // information at the cost of one extra rounding.
684 if(rhs.lower() == value_t(0) || rhs.upper() == value_t(0)) {
685 return *this *= reciprocal_(rhs);
686 }
687
688 auto quotient = m_closure_ / rhs.m_closure_;
689
690 auto ll = lower() / rhs.lower();
691 auto lh = lower() / rhs.upper();
692 auto hl = upper() / rhs.lower();
693 auto hh = upper() / rhs.upper();
694
695 // A quotient is zero exactly when its dividend is
696 const bool zero_attained = contains(value_t(0));
697 return *this =
698 from_corners_(quotient, ll, lh, hl, hh, *this, rhs, zero_attained);
699}
