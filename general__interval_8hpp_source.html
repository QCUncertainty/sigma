501template<typename ValueType>
502bool GeneralInterval<ValueType>::contains(const GeneralInterval& other) const {
503 if(other.empty()) { return true; }
504 if(!m_closure_.contains(other.m_closure_)) { return false; }
505
506 // *this and other may share a bound. For each shared bound the choices
507 // are:
508 // *this open and other closed -> not okay
509 // *this open and other open -> okay
510 // *this closed and other closed -> okay
511 // *this closed and other open -> okay
512 if(other.lower() == lower()) {
513 if(left_open() && other.left_closed()) { return false; }
514 }
515 if(other.upper() == upper()) {
516 if(right_open() && other.right_closed()) { return false; }
517 }
518 return true;
519}

521template<typename ValueType>
522GeneralInterval<ValueType> GeneralInterval<ValueType>::set_union(
523 const GeneralInterval& other) const {
524 if(empty()) { return other; }
525 if(other.empty()) { return *this; }
526 // The intervals need not share a point: [0, 1) and [1, 2] only touch, but
527 // their union is [0, 2]. What breaks the union is a gap, i.e., bounds that
528 // are separated or that coincide at a point neither interval contains.
529 const auto separated = [](const GeneralInterval& left,
530 const GeneralInterval& right) {
531 if(left.upper() < right.lower()) { return true; }
532 return left.upper() == right.lower() && left.right_open() &&
533 right.left_open();
534 };
535 if(separated(*this, other) || separated(other, *this)) {
536 throw std::domain_error("Intervals do not overlap");
537 }
538 value_t low = lower();
539 value_t hi = upper();
540 bool lower_is_open = false;
541 bool upper_is_open = false;
542 if(lower() < other.lower()) { // *this has the lower bound
543 lower_is_open = left_open();
544 } else if(lower() == other.lower()) {
545 // If both are open, then result is open; otherwise closed
546 lower_is_open = left_open() && other.left_open();
547 } else {
548 low = other.lower();
549 lower_is_open = other.left_open();
550 }
551
552 if(upper() > other.upper()) {
553 upper_is_open = right_open();
554 } else if(upper() == other.upper()) {
555 upper_is_open = right_open() && other.right_open();
556 } else {
557 hi = other.upper();
558 upper_is_open = other.right_open();
559 }
560
561 return GeneralInterval(low, hi, lower_is_open, upper_is_open);
562}

564template<typename ValueType>
565GeneralInterval<ValueType> GeneralInterval<ValueType>::set_intersection(
566 const GeneralInterval& other) const {

568 if(other.upper() < lower() || other.lower() > upper()) {

604template<typename ValueType>
605bool GeneralInterval<ValueType>::corner_bound_open_(
606 value_t value, value_t ll, value_t lh, value_t hl, value_t hh,
607 const GeneralInterval& lhs, const GeneralInterval& rhs) {
608 bool is_open = true;
609 if(value == ll) { is_open &= lhs.left_open() || rhs.left_open(); }
610 if(value == lh) { is_open &= lhs.left_open() || rhs.right_open(); }
611 if(value == hl) { is_open &= lhs.right_open() || rhs.left_open(); }
612 if(value == hh) { is_open &= lhs.right_open() || rhs.right_open(); }
613 return is_open;
614}

616template<typename ValueType>
617GeneralInterval<ValueType> GeneralInterval<ValueType>::from_corners_(
618 const closure_t& result, value_t ll, value_t lh, value_t hl, value_t hh,
619 const GeneralInterval& lhs, const GeneralInterval& rhs, bool zero_attained) {
620 // fmin and fmax skip the NaN of a zero times an infinite bound, which is
621 // not a corner the result can come from.
622 auto low = std::fmin(std::fmin(ll, lh), std::fmin(hl, hh));
623 auto high = std::fmax(std::fmax(ll, lh), std::fmax(hl, hh));
624
625 // An attained zero is checked against the result rather than the corners,
626 // which are all NaN for, e.g., [0, 0] * (-inf, inf).
627 bool lower_is_open = !(zero_attained && result.lower() == value_t(0)) &&
628 corner_bound_open_(low, ll, lh, hl, hh, lhs, rhs);
629 bool upper_is_open = !(zero_attained && result.upper() == value_t(0)) &&
630 corner_bound_open_(high, ll, lh, hl, hh, lhs, rhs);
631
632 return GeneralInterval(result.lower(), result.upper(), lower_is_open,
633 upper_is_open);
634}

653template<typename ValueType>
654GeneralInterval<ValueType> GeneralInterval<ValueType>::reciprocal_(
655 const GeneralInterval& a) {
656 using interval_t = detail_::boost_interval_t<value_t>;
657 constexpr auto inf = std::numeric_limits<value_t>::infinity();
658 const interval_t one(value_t(1));
659
660 // A zero bound is necessarily an open one here, and the reciprocal runs
661 // off to infinity in that direction. Everything else is a directed
662 // division by a point.
663 value_t low =
664 a.upper() == value_t(0) ? -inf : (one / interval_t(a.upper())).lower();
665 value_t high =
666 a.lower() == value_t(0) ? inf : (one / interval_t(a.lower())).upper();
667
668 // The reciprocal is decreasing, so the bounds trade places
669 return GeneralInterval(low, high, a.right_open(), a.left_open());
670}

672template<typename ValueType>
673GeneralInterval<ValueType>& GeneralInterval<ValueType>::operator/=(
674 const GeneralInterval& rhs) {
675 if(empty() || rhs.empty()) { return *this = GeneralInterval(); }
676 if(rhs.contains(value_t(0))) {
677 throw std::domain_error("Division by zero");
678 }
679
680 // The closure of a divisor bounded by an open zero contains zero, so
681 // closed division would reject it. Multiplying by the reciprocal keeps the
682 // information at the cost of one extra rounding.
683 if(rhs.lower() == value_t(0) || rhs.upper() == value_t(0)) {
684 return *this *= reciprocal_(rhs);
685 }
686
687 auto quotient = m_closure_ / rhs.m_closure_;
688
689 auto ll = lower() / rhs.lower();
690 auto lh = lower() / rhs.upper();
691 auto hl = upper() / rhs.lower();
692 auto hh = upper() / rhs.upper();
693
694 // A quotient is zero exactly when its dividend is
695 const bool zero_attained = contains(value_t(0));
696 return *this =
697 from_corners_(quotient, ll, lh, hl, hh, *this, rhs, zero_attained);
698}
