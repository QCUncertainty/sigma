794template<typename ValueType>
795auto TaylorModel<ValueType>::operator+=(const TaylorModel& other)
796 -> TaylorModel& {
797 if(empty()) { return *this = other; }
798 if(other.empty()) { return *this; }
799 auto new_order = std::min(max_order(), other.max_order());
800 auto lhs = sweep_to_order(new_order);
801 auto rhs = other.sweep_to_order(new_order);
802 return *this = TaylorModel(lhs.m_polynomial_ + rhs.m_polynomial_,
803 lhs.m_remainder_ + rhs.m_remainder_);
804}

806template<typename ValueType>
807auto TaylorModel<ValueType>::operator*=(const TaylorModel& other)
808 -> TaylorModel& {
809 if(empty() || other.empty()) { return *this = TaylorModel(); }
810 auto new_order = std::min(max_order(), other.max_order());
811 auto lhs = sweep_to_order(new_order);
812 auto rhs = other.sweep_to_order(new_order);
813
814 const auto& p1 = lhs.m_polynomial_;
815 const auto& p2 = rhs.m_polynomial_;
816
817 value_t new_constant = p1.constant() * p2.constant();
818 coeffs_t kept, tail;
819 auto insert = [&](const monomial_t& mono, value_t coeff) {
820 if(mono.degree() <= new_order) {
821 kept[mono] += coeff;
822 } else {
823 tail[mono] += coeff;
824 }
825 };
826 for(auto&& [mono, coeff] : p2.coefficients()) {
827 insert(mono, p1.constant() * coeff);
828 }
829 for(auto&& [mono, coeff] : p1.coefficients()) {
830 insert(mono, coeff * p2.constant());
831 }
832 for(auto&& [mono1, coeff1] : p1.coefficients()) {
833 for(auto&& [mono2, coeff2] : p2.coefficients()) {
834 insert(mono1 * mono2, coeff1 * coeff2);
835 }
836 }
837
838 // Every term of tail has degree <= order(p1) + order(p2) == 2*new_order.
839 taylor_t tail_poly(value_t{0}, std::move(tail), Order(2 * new_order));
840 interval_t new_remainder =
841 tail_poly.bound() + p1.bound() * rhs.m_remainder_ +
842 p2.bound() * lhs.m_remainder_ + lhs.m_remainder_ * rhs.m_remainder_;
843 taylor_t new_poly(new_constant, std::move(kept), Order(new_order));
844 return *this = TaylorModel(std::move(new_poly), std::move(new_remainder));
845}

847template<typename ValueType>
848auto TaylorModel<ValueType>::compose_(
849 const std::vector<value_t>& outer_coeffs,
850 const std::function<interval_t(const interval_t&)>& nth_derivative) const
851 -> TaylorModel {
852 assert_not_empty_();
853 auto order = max_order();
854 taylor_t new_poly = m_polynomial_.compose_(outer_coeffs);
855 interval_t bar_bound = m_polynomial_.bound() - m_polynomial_.constant();
856 interval_t enclosure = bar_bound + m_remainder_;
857 interval_t xi_range =
858 m_polynomial_.constant() + interval_t(value_t{0}, value_t{1}) * enclosure;
859 interval_t deriv_val = nth_derivative(xi_range);
860 value_t fact = detail::factorial<value_t>(order + 1);
861 interval_t new_remainder =
862 deriv_val * pow(enclosure, static_cast<int>(order + 1)) / fact;
863 return TaylorModel(std::move(new_poly), std::move(new_remainder));
864}

866template<typename ValueType>
867auto TaylorModel<ValueType>::multiplicative_inverse() const -> TaylorModel {
868 assert_not_empty_();
869 if(range().contains(value_t{0})) {
870 throw std::domain_error("Division by zero");
871 }
872 auto order = max_order();
873 auto outer_coeffs =
874 detail::reciprocal_outer_coeffs(m_polynomial_.constant(), order);
875 return compose_(outer_coeffs, [order](const interval_t& xi) {
876 value_t sign = ((order + 1) % 2 == 0) ? value_t{1} : value_t{-1};
877 return sign * detail::factorial<value_t>(order + 1) /
878 pow(xi, static_cast<int>(order + 2));
879 });

882template<typename ValueType>
883auto TaylorModel<ValueType>::sweep_to_order(size_type new_order) const
884 -> TaylorModel {
885 assert_not_empty_();
886 coeffs_t kept, tail;
887 for(auto&& [mono, coeff] : m_polynomial_.coefficients()) {
888 if(mono.degree() <= new_order) {
889 kept[mono] = coeff;
890 } else {
891 tail[mono] = coeff;
892 }
893 }
894 taylor_t tail_poly(value_t{0}, std::move(tail),
895 Order(m_polynomial_.max_order()));
896 auto new_remainder = m_remainder_ + tail_poly.bound();
897 taylor_t new_poly(m_polynomial_.constant(), std::move(kept),
898 Order(new_order));
899 return TaylorModel(std::move(new_poly), std::move(new_remainder));
900}

902template<typename ValueType>
903auto TaylorModel<ValueType>::sweep_small(value_t threshold) const
904 -> TaylorModel {
905 assert_not_empty_();
906 auto total_r = range().radius();
907 coeffs_t kept, swept;
908 for(auto&& [mono, coeff] : m_polynomial_.coefficients()) {
909 if(total_r != value_t{0} && std::fabs(coeff) / total_r < threshold) {
910 swept[mono] = coeff;
911 } else {
912 kept[mono] = coeff;
913 }
914 }
915 taylor_t swept_poly(value_t{0}, std::move(swept),
916 Order(m_polynomial_.max_order()));
917 auto new_remainder = m_remainder_ + swept_poly.bound();
918 taylor_t new_poly(m_polynomial_.constant(), std::move(kept),
919 Order(m_polynomial_.max_order()));
920 return TaylorModel(std::move(new_poly), std::move(new_remainder));
921}

949template<typename T>
950TaylorModel<T> abs(const TaylorModel<T>& a) {
951 if(a.empty()) { return a; }
952 auto a_range = a.range();
953 if(a_range.lower() >= T(0.0)) { return a; }
954 if(a_range.upper() <= T(0.0)) { return -a; }
955 auto m = std::max(std::fabs(a_range.lower()), a_range.upper());
956 return TaylorModel<T>(typename TaylorModel<T>::interval_t(T(0.0), m),
957 typename TaylorModel<T>::Order(a.max_order()));
958}
