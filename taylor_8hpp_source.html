733 static interval_t diagonal_extrema_(value_t lin, value_t quad) {
734 if(quad == value_t{0}) {
735 auto mag = std::fabs(lin);
736 return interval_t(-mag, mag);
737 }
738 value_t f_lo = quad - lin; // f(-1)
739 value_t f_hi = quad + lin; // f(1)
740 value_t vertex = -lin / (value_t{2} * quad);
741 value_t f_vertex = -(lin * lin) / (value_t{4} * quad);
742 bool vertex_in_domain = vertex >= value_t{-1} && vertex <= value_t{1};
743
744 if(quad > value_t{0}) { // convex: vertex is the minimum
745 auto lo = vertex_in_domain ? f_vertex : std::min(f_lo, f_hi);
746 return interval_t(lo, std::max(f_lo, f_hi));
747 }
748 // concave: vertex is the maximum
749 auto hi = vertex_in_domain ? f_vertex : std::max(f_lo, f_hi);
750 return interval_t(std::min(f_lo, f_hi), hi);
751 }

818template<typename ValueType>
819auto Taylor<ValueType>::bound() const -> interval_t {
820 assert_not_empty_();
821 interval_t domain(value_t{-1}, value_t{1});
822
823 // Each variable's linear and pure-quadratic coefficients (c_{e_i} and
824 // c_{2e_i}), collected so their sum can be bounded exactly below.
825 std::map<deviation_t, std::pair<value_t, value_t>> diagonal;
826
827 // Naive per-term bound of everything QFB doesn't capture exactly: cross
828 // terms (more than one variable) and degree->= 3 single-variable terms.
829 interval_t remainder(value_t{0}, value_t{0});
830
831 for(auto&& [mono, coeff] : m_coeffs_) {
832 auto&& exps = mono.exponents();
833 if(exps.size() == 1) {
834 auto [var, exp] = *exps.begin();
835 if(exp == 1) {
836 diagonal[var].first += coeff;
837 continue;
838 }
839 if(exp == 2) {
840 diagonal[var].second += coeff;
841 continue;
842 }
843 }
844 interval_t term(value_t{1}, value_t{1});
845 for(auto&& [var, exp] : exps) { term *= pow(domain, exp); }
846 term *= coeff;
847 remainder += term;
848 }
849
850 interval_t total(constant(), constant());
851 total += remainder;
852 for(auto&& [var, coeffs] : diagonal) {
853 auto [lin, quad] = coeffs;
854 total += diagonal_extrema_(lin, quad);
855 }
856 return total;
857}

859template<typename ValueType>
860auto Taylor<ValueType>::derivative(deviation_t var) const -> Taylor {
861 assert_not_empty_();
862 value_t new_constant{0};
863 coeffs_t new_coeffs;
864 for(auto&& [mono, coeff] : m_coeffs_) {
865 auto e = mono.exponent(var);
866 if(e == 0) { continue; }
867 auto exps = mono.exponents();
868 exps[var] = e - 1;
869 value_t new_coeff = coeff * static_cast<value_t>(e);
870 monomial_t new_mono(exps);
871 if(new_mono.empty()) {
872 new_constant += new_coeff;
873 } else {
874 new_coeffs[new_mono] += new_coeff;
875 }
876 }
877 return Taylor(new_constant, std::move(new_coeffs), Order(m_order_));
878}

923template<typename ValueType>
924auto Taylor<ValueType>::operator*=(const Taylor& other) -> Taylor& {
925 if(empty() || other.empty()) { return *this = Taylor(); }
926 auto new_order = std::min(m_order_, other.m_order_);
927 auto lhs = truncate(new_order);
928 auto rhs = other.truncate(new_order);
929 value_t new_constant = (*lhs.m_constant_) * (*rhs.m_constant_);
930 coeffs_t new_coeffs;
931 for(auto&& [mono, coeff] : rhs.m_coeffs_) {
932 new_coeffs[mono] += (*lhs.m_constant_) * coeff;
933 }
934 for(auto&& [mono, coeff] : lhs.m_coeffs_) {
935 new_coeffs[mono] += coeff * (*rhs.m_constant_);
936 }
937 for(auto&& [mono1, c1] : lhs.m_coeffs_) {
938 for(auto&& [mono2, c2] : rhs.m_coeffs_) {
939 auto mono = mono1 * mono2;
940 if(mono.degree() > new_order) { continue; }
941 new_coeffs[mono] += c1 * c2;
942 }
943 }
944 return *this =
945 Taylor(new_constant, std::move(new_coeffs), Order(new_order));
946}

948template<typename ValueType>
949auto Taylor<ValueType>::compose_(const std::vector<value_t>& outer_coeffs) const
950 -> Taylor {
951 assert_not_empty_();
952 Taylor bar(value_t{0}, m_coeffs_, Order(m_order_));
953 value_t c0 = outer_coeffs.empty() ? value_t{0} : outer_coeffs[0];
954 Taylor result = Taylor(c0, Order(m_order_));
955 Taylor power = bar;
956 for(size_type k = 1; k < outer_coeffs.size(); ++k) {
957 result += power * outer_coeffs[k];
958 if(k + 1 < outer_coeffs.size()) { power *= bar; }
959 }
960 return result;
961}
