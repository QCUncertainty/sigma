15template<typename T>
16Interval<T> cosh(const Interval<T>& a) {
17 if(a.empty()) { return Interval<T>(); }
18
19 bool left_open = a.left_open();
20 bool right_open = a.right_open();
21 if(a.upper() < T(0)) {
22 // Entirely negative, where cosh is decreasing
23 std::swap(left_open, right_open);
24 } else if(a.lower() < T(0)) {
25 // Spans zero, so the minimum is the interior point cosh(0) == 1 and
26 // the maximum comes from whichever bound is farther from zero.
27 const T abs_lower = -a.lower();
28 left_open = false;
29 if(abs_lower > a.upper()) {
30 right_open = a.left_open();
31 } else if(abs_lower < a.upper()) {
32 right_open = a.right_open();
33 } else {
34 right_open = a.left_open() && a.right_open();
35 }
36 }
37
38 return detail_::from_boost_(boost::numeric::cosh(detail_::to_boost_(a)),
39 left_open, right_open);
40}

56template<typename T>
57Interval<T> acosh(const Interval<T>& a) {
58 if(a.empty()) { return Interval<T>(); }
59 // Trimming the argument back to the domain would answer a narrower
60 // question than the one asked. Boost would clamp the lower bound to one.
61 if(a.lower() < T(1)) {
62 throw std::domain_error("Interval has values less than 1.");
63 }
64 // Increasing, so the openness of each bound carries over.
65 return detail_::from_boost_(boost::numeric::acosh(detail_::to_boost_(a)),
66 a.left_open(), a.right_open());
67}

69template<typename T>
70Interval<T> atanh(const Interval<T>& a) {
71 if(a.empty()) { return Interval<T>(); }
72 // The domain is open, so a bound AT plus or minus one is out of domain
73 // only if it is closed -- the same rule log() applies at zero.
74 detail_::assert_within_unit_(a, true, "inverse hyperbolic tangent");
75 // Increasing. An open bound at either end of the domain leaves the result
76 // unbounded on that side, and from_boost_ opens that infinite bound.
77 return detail_::from_boost_(boost::numeric::atanh(detail_::to_boost_(a)),
78 a.left_open(), a.right_open());
79}
