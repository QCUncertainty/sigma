22template<typename T>
23Interval<T> tan(const Interval<T>& a) {
24 if(a.empty()) { return Interval<T>(); }
25
26 auto y = boost::numeric::tan(detail_::to_boost_(a));
27
28 // Boost reduces the argument against the period and reports "this reaches
29 // a pole" by returning the whole real line, which is the only way it
30 // produces an unbounded result: every other path bounds the argument
31 // strictly inside a single branch, where the tangent is finite. That makes
32 // the test below exactly boost's own pole detection, with no period
33 // arithmetic of ours to get wrong.
34 //
35 // The detection is conservative, as it must be: an odd multiple of pi/2 is
36 // not representable, so for an argument whose bound sits within rounding
37 // distance of one, the reduction cannot certify that the pole is excluded.
38 // Such an argument is refused rather than answered.
39 if(std::isinf(y.lower()) && std::isinf(y.upper())) {
40 throw std::domain_error("Interval contains an odd multiple of pi/2, "
41 "where the tangent is undefined.");
42 }
43
44 // Every remaining argument lies strictly inside one branch, on which the
45 // tangent is increasing, so the openness of each bound carries over.
46 return detail_::from_boost_(y, a.left_open(), a.right_open());
47}

49template<typename T>
50Interval<T> asin(const Interval<T>& a) {
51 if(a.empty()) { return Interval<T>(); }
52 // Trimming the argument back to the domain would answer a narrower
53 // question than the one asked. Boost would do exactly that.
54 detail_::assert_within_unit_(a, false, "arcsine");
55 // Increasing, so the openness of each bound carries over.
56 return detail_::from_boost_(boost::numeric::asin(detail_::to_boost_(a)),
57 a.left_open(), a.right_open());
58}
