20template<typename T>
21Interval<T> tan(const Interval<T>& a) {
22 if(a.empty()) { return Interval<T>(); }
23
24 auto y = boost::numeric::tan(detail_::to_boost_(a));
25
26 // Boost reduces the argument against the period and reports "this reaches
27 // a pole" by returning the whole real line, which is the only way it
28 // produces an unbounded result: every other path bounds the argument
29 // strictly inside a single branch, where the tangent is finite. That makes
30 // the test below exactly boost's own pole detection, with no period
31 // arithmetic of ours to get wrong.
32 //
33 // The detection is conservative, as it must be: an odd multiple of pi/2 is
34 // not representable, so for an argument whose bound sits within rounding
35 // distance of one, the reduction cannot certify that the pole is excluded.
36 // Such an argument is refused rather than answered.
37 if(std::isinf(y.lower()) && std::isinf(y.upper())) {
38 throw std::domain_error("Interval contains an odd multiple of pi/2, "
39 "where the tangent is undefined.");
40 }
41
42 return detail_::from_boost_(y);
43}
